88 DecodableMatrixScaledMapped(const TransitionInformation &tm,
89 const FloatMatrix &likes, float scale)

99 }

105 DecodableMatrixScaledMapped(const TransitionInformation &tm, float scale,
106 const FloatMatrix *likes)
107 : trans_model_(tm),
108 likes_(likes),
109 tid_to_pdf_(trans_model_.TransitionIdToPdfArray()),
110 scale_(scale),
111 delete_likes_(true) {
112 if (likes->cols() != tm.NumPdfs())
113 KALDI_DECODER_ERR << "DecodableMatrixScaledMapped: mismatch, matrix has "
114 << likes->cols() << " cols but transition-model has "
115 << tm.NumPdfs() << " pdf-ids.";
116 }
117
119 int32_t NumFramesReady() const override { return likes_->rows(); }
120
121 bool IsLastFrame(int32_t frame) const override {
122 KALDI_DECODER_ASSERT(frame < NumFramesReady());
123 return frame == NumFramesReady() - 1;
124 }
125
130 float LogLikelihood(int32_t frame, int32_t tid) override {
131 KALDI_DECODER_PARANOID_ASSERT(tid >= 1 &&
132 tid < static_cast<int32_t>(tid_to_pdf_.size()));
133 KALDI_DECODER_PARANOID_ASSERT(frame >= 0 && frame < NumFramesReady());
134 return scale_ * (*likes_)(frame, tid_to_pdf_[tid]);
135 }
136
138 int32_t NumIndices() const override { return trans_model_.NumTransitionIds(); }
139
141 ~DecodableMatrixScaledMapped() override {
142 if (delete_likes_) delete likes_;
143 }
144
145 private:
147 const TransitionInformation &trans_model_;
149 const FloatMatrix *likes_;
151 const std::vector<int32_t> &tid_to_pdf_;
153 float scale_;
155 bool delete_likes_;
156 KALDI_DECODER_DISALLOW_COPY_AND_ASSIGN(DecodableMatrixScaledMapped);
157};

233 tid_to_pdf_(trans_model_.TransitionIdToPdfArray()),
234 frame_offset_(0),
235 input_is_finished_(false) {}
236
238 int32_t FirstAvailableFrame() const { return frame_offset_; }
239
246 void AcceptLoglikes(FloatMatrix *loglikes, int32_t frames_to_discard);
247
248 void InputIsFinished() { input_is_finished_ = true; }
249
250 int32_t NumFramesReady() const override {
251 return loglikes_.rows() + frame_offset_;
252 }
253
254 bool IsLastFrame(int32_t frame) const override {
255 KALDI_DECODER_ASSERT(frame < NumFramesReady());
256 return frame == NumFramesReady() - 1 && input_is_finished_;
257 }
258
259 float LogLikelihood(int32_t frame, int32_t tid) override {
260 KALDI_DECODER_PARANOID_ASSERT(tid >= 1 &&
261 tid < static_cast<int32_t>(tid_to_pdf_.size()));
262 KALDI_DECODER_PARANOID_ASSERT(frame >= frame_offset_ &&
263 frame < NumFramesReady());
264 int32_t pdf_id = tid_to_pdf_[tid];
265 return loglikes_(frame - frame_offset_, pdf_id);
266 }
267
268 // Note: these indices are 1-based.
269 int32_t NumIndices() const override { return trans_model_.NumTransitionIds(); }
270
271 // nothing special to do in destructor.
272 ~DecodableMatrixMappedOffset() override = default;
273
274 private:
276 const TransitionInformation &trans_model_;
278 const std::vector<int32_t> &tid_to_pdf_;
280 FloatMatrix loglikes_;
282 int32_t frame_offset_;
284 bool input_is_finished_;
285 KALDI_DECODER_DISALLOW_COPY_AND_ASSIGN(DecodableMatrixMappedOffset);
286};

293class DecodableMatrixScaled : public DecodableInterface {
294 public:
295 DecodableMatrixScaled(const FloatMatrix &likes, float scale)
296 : likes_(likes), scale_(scale) {}
297
298 int32_t NumFramesReady() const override { return likes_.rows(); }
299
300 bool IsLastFrame(int32_t frame) const override {
301 KALDI_DECODER_ASSERT(frame < NumFramesReady());
302 return frame == NumFramesReady() - 1;
303 }
304
305 // Note, frames are numbered from zero.
306 float LogLikelihood(int32_t frame, int32_t index) override {
307 if (index > likes_.cols() || index <= 0 || frame < 0 ||
308 frame >= likes_.rows())
309 KALDI_DECODER_ERR << "Invalid (frame, index - 1) = (" << frame << ", "
310 << index - 1 << ") for matrix of size " << likes_.rows()
311 << " x " << likes_.cols();
312 return scale_ * likes_(frame, index - 1);
313 }
314
315 // Indices are one-based! This is for compatibility with OpenFst.
316 int32_t NumIndices() const override { return likes_.cols(); }
317
318 private:
320 const FloatMatrix &likes_;
322 float scale_;
323 KALDI_DECODER_DISALLOW_COPY_AND_ASSIGN(DecodableMatrixScaled);
324};
